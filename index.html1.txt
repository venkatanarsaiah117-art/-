        let cart = [];

        function renderProducts() {
            const grid = document.getElementById('products-grid');
            grid.innerHTML = '';
            products.forEach((p, index) => {
                let optionsHTML = `<div class="options-group"><label>పరిమాణం / ప్యాక్ ఎంచుకోండి:</label><select id="opt-${index}" onchange="updatePrice(${index})">`;
                p.options.forEach((opt, optIdx) => {
                    optionsHTML += `<option value="${optIdx}">${opt.label} - ₹${opt.price}</option>`;
                });
                optionsHTML += `</select></div>`;

                grid.innerHTML += `
                    <div class="product-card">
                        <div class="product-img-wrapper">
                            <img src="${p.img}" class="product-img" alt="${p.name}">
                        </div>
                        <div class="product-info">
                            <div class="product-title">${p.name}</div>
                            <div class="product-desc">${p.desc}</div>
                            <div class="product-benefits"><strong>💡 ఆరోగ్య లాభాలు:</strong> ${p.benefits}</div>
                            ${optionsHTML}
                            <div class="price-row">
                                <span class="price" id="price-${index}">₹${p.options[0].price}</span>
                                <button class="add-btn" onclick="addToCart(${index})">కార్ట్‌కి జోడించు</button>
                            </div>
                        </div>
                    </div>
                `;
            });
        }

        function updatePrice(index) {
            const select = document.getElementById(`opt-${index}`);
            const selectedOptIdx = select.value;
            const price = products[index].options[selectedOptIdx].price;
            document.getElementById(`price-${index}`).innerText = `₹${price}`;
        }

        function addToCart(index) {
            const select = document.getElementById(`opt-${index}`);
            const selectedOptIdx = select.value;
            const selectedOption = products[index].options[selectedOptIdx];

            const cartItem = {
                name: products[index].name,
                size: selectedOption.label,
                price: selectedOption.price
            };

            cart.push(cartItem);
            updateCartUI();
        }

        function removeFromCart(cartIndex) {
            cart.splice(cartIndex, 1);
            updateCartUI();
        }

        function updateCartUI() {
            const cartList = document.getElementById('cart-items');
            const totalEl = document.getElementById('cart-total');

            if (cart.length === 0) {
                cartList.innerHTML = `<p style="color: #777; text-align: center;">మీ కార్ట్ ఖాళీగా ఉంది. కావాల్సిన ప్రొడక్ట్స్ ని యాడ్ చేయండి!</p>`;
                totalEl.innerText = `₹0`;
                return;
            }

            cartList.innerHTML = '';
            let total = 0;

            cart.forEach((item, idx) => {
                total += item.price;
                cartList.innerHTML += `
                    <li class="cart-item">
                        <span>${item.name} (${item.size})</span>
                        <span><strong>₹${item.price}</strong> <button class="remove-btn" onclick="removeFromCart(${idx})">❌</button></span>
                    </li>
                `;
            });

            totalEl.innerText = `₹${total}`;
        }
